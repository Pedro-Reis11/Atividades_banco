/* =============================================================================
   1) JOINS BÁSICOS (banco empresa)
   ============================================================================= */
USE empresa;

-- -----------------------------------------------------------------------------
-- Questão 1
-- Liste o nome do fornecedor e o nome do produto para todos os fornecedores
-- que possuem produtos cadastrados.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto
FROM Fornecedor f
INNER JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor;

-- -----------------------------------------------------------------------------
-- Questão 2
-- Mostre todos os fornecedores e, se existirem, os produtos relacionados.
-- Inclua também os fornecedores sem produto.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto
FROM Fornecedor f
LEFT JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor;

-- -----------------------------------------------------------------------------
-- Questão 3
-- Exiba todos os produtos e, quando houver, o nome do fornecedor correspondente.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto
FROM Fornecedor f
RIGHT JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor;

-- -----------------------------------------------------------------------------
-- Questão 4
-- Verifique quais fornecedores não possuem nenhum produto cadastrado.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       f.cnpj,
       p.nome AS produto
FROM Fornecedor f
LEFT JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor
WHERE p.cnpj_fornecedor IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 5
-- Apresente os fornecedores, seus produtos e a identificação dos produtos.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.id,
       p.nome AS produto
FROM Fornecedor f
INNER JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor;


/* =============================================================================
   2) JOINS COM FILTROS E ORDENAÇÃO (banco empresa)
   ============================================================================= */

-- -----------------------------------------------------------------------------
-- Questão 6
-- Liste somente os produtos cujo preço seja maior que R$ 50,00 junto com o
-- nome do fornecedor.
-- -----------------------------------------------------------------------------
SELECT e.preco,
       p.nome AS produto,
       f.nome AS fornecedor
FROM Estoque e
INNER JOIN Produto p
    ON e.id_produto = p.id
INNER JOIN Fornecedor f
    ON p.cnpj_fornecedor = f.cnpj
WHERE e.preco > 50
ORDER BY f.nome, p.nome;

-- -----------------------------------------------------------------------------
-- Questão 7
-- Mostre os fornecedores cujo nome começa com "A" e os produtos relacionados,
-- ordenados por fornecedor e produto.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto
FROM Fornecedor f
INNER JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor
WHERE f.nome LIKE 'A%'
ORDER BY f.nome, p.nome;

-- -----------------------------------------------------------------------------
-- Questão 8
-- Exiba os fornecedores e produtos em ordem alfabética decrescente por
-- fornecedor.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto
FROM Fornecedor f
LEFT JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor
ORDER BY f.nome DESC;

-- -----------------------------------------------------------------------------
-- Questão 9
-- Liste os fornecedores com seus produtos, mas somente quando o produto tiver
-- preço entre R$ 20,00 e R$ 100,00.
-- -----------------------------------------------------------------------------
SELECT f.nome AS fornecedor,
       p.nome AS produto,
       e.preco
FROM Fornecedor f
INNER JOIN Produto p
    ON f.cnpj = p.cnpj_fornecedor
INNER JOIN Estoque e
    ON e.id_produto = p.id
WHERE e.preco BETWEEN 20 AND 100;

-- -----------------------------------------------------------------------------
-- Questão 10
-- Retorne os nomes dos fornecedores, os nomes dos produtos desses fornecedores,
-- a quantidade desses produtos no estoque e as filiais em que são vendidos.
-- Inclua também os fornecedores sem produtos.
-- -----------------------------------------------------------------------------
SELECT p.nome AS produto,
       f.nome AS fornecedor,
       e.quantidade,
       Filial.nome AS filial
FROM Estoque e
INNER JOIN Produto p
    ON e.id_produto = p.id
RIGHT JOIN Fornecedor f
    ON p.cnpj_fornecedor = f.cnpj
LEFT JOIN Filial
    ON e.cnpj_filial = Filial.cnpj;

);

CREATE TABLE Pedido (
    id_pedido   INT PRIMARY KEY,
    id_cliente  INT NOT NULL,
    data_pedido DATE NOT NULL,
    total       DECIMAL(10,2) NOT NULL,
    FOREIGN KEY (id_cliente) REFERENCES Cliente(id_cliente)
);

CREATE TABLE Produto (
    id_produto INT PRIMARY KEY,
    nome       VARCHAR(100) NOT NULL,
    preco      DECIMAL(10,2) NOT NULL
);

CREATE TABLE ItemPedido (
    id_pedido  INT NOT NULL,
    id_produto INT NOT NULL,
    quantidade INT NOT NULL,
    PRIMARY KEY (id_pedido, id_produto),
    FOREIGN KEY (id_pedido)  REFERENCES Pedido(id_pedido),
    FOREIGN KEY (id_produto) REFERENCES Produto(id_produto)
);

-- -----------------------------------------------------------------------------
-- Questão 11
-- Liste todos os clientes e os pedidos realizados por eles, incluindo clientes
-- que ainda não fizeram pedidos.
-- -----------------------------------------------------------------------------
SELECT c.nome AS cliente,
       p.id_pedido,
       p.data_pedido,
       p.total
FROM Cliente c
LEFT JOIN Pedido p
    ON p.id_cliente = c.id_cliente;

-- -----------------------------------------------------------------------------
-- Questão 12
-- Mostre cada pedido com o nome do cliente e o valor total do pedido.
-- -----------------------------------------------------------------------------
SELECT c.nome AS cliente,
       p.id_pedido,
       p.total
FROM Pedido p
INNER JOIN Cliente c
    ON c.id_cliente = p.id_cliente;

-- -----------------------------------------------------------------------------
-- Questão 13
-- Liste todos os produtos vendidos em cada pedido, com o nome do produto,
-- quantidade e valor total do item.
-- -----------------------------------------------------------------------------
SELECT pe.id_pedido,
       p.nome AS produto,
       ip.quantidade,
       p.preco AS preco_unitario,
       p.preco * ip.quantidade AS total_item
FROM Pedido pe
INNER JOIN ItemPedido ip
    ON pe.id_pedido = ip.id_pedido
INNER JOIN Produto p
    ON ip.id_produto = p.id_produto
ORDER BY pe.id_pedido, p.nome;

-- -----------------------------------------------------------------------------
-- Questão 14
-- Descubra quais clientes nunca fizeram pedido.
-- -----------------------------------------------------------------------------
SELECT c.id_cliente,
       c.nome AS cliente
FROM Cliente c
LEFT JOIN Pedido p
    ON p.id_cliente = c.id_cliente
WHERE p.id_pedido IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 15
-- Mostre o nome do cliente, o número do pedido e o total do pedido,
-- considerando apenas pedidos com valor acima de R$ 80,00.
-- -----------------------------------------------------------------------------
SELECT c.nome AS cliente,
       p.id_pedido,
       p.total
FROM Cliente c
INNER JOIN Pedido p
    ON p.id_cliente = c.id_cliente
WHERE p.total > 80;

-- -----------------------------------------------------------------------------
-- Questão 16
-- Liste todos os clientes e suas reservas, incluindo clientes sem reservas.
-- -----------------------------------------------------------------------------
SELECT c.nome AS cliente,
       r.id_reserva,
       r.data_viagem,
       r.valor
FROM Cliente c
LEFT JOIN Reserva r
    ON r.id_cliente = c.id_cliente;

-- -----------------------------------------------------------------------------
-- Questão 17
-- Mostre cada reserva com o nome do cliente e o nome do destino.
-- -----------------------------------------------------------------------------
SELECT r.id_reserva,
       c.nome AS cliente,
       d.nome_destino AS destino,
       r.data_viagem,
       r.valor
FROM Reserva r
INNER JOIN Cliente c
    ON c.id_cliente = r.id_cliente
INNER JOIN Destino d
    ON d.id_destino = r.id_destino;

-- -----------------------------------------------------------------------------
-- Questão 18
-- Exiba os destinos que ainda não receberam reserva.
-- -----------------------------------------------------------------------------
SELECT d.id_destino,
       d.nome_destino,
       d.pais
FROM Destino d
LEFT JOIN Reserva r
    ON r.id_destino = d.id_destino
WHERE r.id_reserva IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 19
-- Exiba os clientes que ainda não fizeram reserva.
-- -----------------------------------------------------------------------------
SELECT c.id_cliente,
       c.nome AS cliente
FROM Cliente c
LEFT JOIN Reserva r
    ON r.id_cliente = c.id_cliente
WHERE r.id_reserva IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 20
-- Exiba todos os clientes e destinos, com ou sem reserva.
-- (O MySQL não tem FULL JOIN: usa-se a união de dois LEFT JOIN.)
-- -----------------------------------------------------------------------------
SELECT c.nome AS cliente,
       d.nome_destino AS destino,
       r.id_reserva,
       r.valor
FROM Cliente c
LEFT JOIN Reserva r
    ON r.id_cliente = c.id_cliente
LEFT JOIN Destino d
    ON d.id_destino = r.id_destino

UNION

SELECT c.nome,
       d.nome_destino,
       r.id_reserva,
       r.valor
FROM Destino d
LEFT JOIN Reserva r
    ON r.id_destino = d.id_destino
LEFT JOIN Cliente c
    ON c.id_cliente = r.id_cliente;

   Tabelas: Medicamento, Fornecedor, Compra

CREATE DATABASE IF NOT EXISTS farmacia;
USE farmacia;

CREATE TABLE Medicamento (
    id_medicamento INT PRIMARY KEY,
    nome           VARCHAR(100) NOT NULL,
    preco          DECIMAL(10,2) NOT NULL,
    validade       DATE NOT NULL,
    laboratorio    VARCHAR(100)
);

CREATE TABLE Fornecedor (
    id_fornecedor INT PRIMARY KEY,
    nome          VARCHAR(100) NOT NULL,
    cidade        VARCHAR(100)
);

CREATE TABLE Compra (
    id_compra      INT PRIMARY KEY,
    id_medicamento INT NOT NULL,
    id_fornecedor  INT NULL,
    quantidade     INT NOT NULL,
    data_compra    DATE NOT NULL,
    FOREIGN KEY (id_medicamento) REFERENCES Medicamento(id_medicamento),
    FOREIGN KEY (id_fornecedor)  REFERENCES Fornecedor(id_fornecedor)

-- -----------------------------------------------------------------------------
-- Questão 21
-- Liste todos os medicamentos com o nome do fornecedor que os forneceu,
-- incluindo medicamentos que ainda não tiveram compra registrada.
-- -----------------------------------------------------------------------------
SELECT m.nome AS medicamento,
       f.nome AS fornecedor
FROM Medicamento m
LEFT JOIN Compra c
    ON c.id_medicamento = m.id_medicamento
LEFT JOIN Fornecedor f
    ON f.id_fornecedor = c.id_fornecedor;

-- -----------------------------------------------------------------------------
-- Questão 22
-- Liste todos os medicamentos sem fornecedores e que foram comprados.
-- -----------------------------------------------------------------------------
SELECT DISTINCT m.nome AS medicamento
FROM Medicamento m
INNER JOIN Compra c
    ON c.id_medicamento = m.id_medicamento
LEFT JOIN Fornecedor f
    ON f.id_fornecedor = c.id_fornecedor
WHERE f.id_fornecedor IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 23
-- Liste todos os medicamentos com fornecedores e que ainda não foram comprados.
-- Obs.: medicamento e fornecedor só se ligam pela tabela Compra; por isso esta
-- consulta lista os fornecedores cadastrados que não têm nenhuma compra.
-- -----------------------------------------------------------------------------
SELECT f.id_fornecedor,
       f.nome AS fornecedor
FROM Fornecedor f
LEFT JOIN Compra c
    ON c.id_fornecedor = f.id_fornecedor
WHERE c.id_compra IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 24
-- Liste todos os medicamentos sem fornecedores e que ainda não foram comprados.
-- -----------------------------------------------------------------------------
SELECT m.id_medicamento,
       m.nome AS medicamento
FROM Medicamento m
LEFT JOIN Compra c
    ON c.id_medicamento = m.id_medicamento
WHERE c.id_compra IS NULL;

-- -----------------------------------------------------------------------------
-- Questão 25
-- Liste todos os dados das compras, dos medicamentos e dos fornecedores, com ou
-- sem registro de fornecedor e de compra. (FULL JOIN emulado com UNION.)
-- -----------------------------------------------------------------------------
SELECT c.id_compra, c.quantidade, c.data_compra,
       m.id_medicamento, m.nome AS medicamento, m.preco, m.validade, m.laboratorio,
       f.id_fornecedor, f.nome AS fornecedor, f.cidade
FROM Medicamento m
LEFT JOIN Compra c
    ON c.id_medicamento = m.id_medicamento
LEFT JOIN Fornecedor f
    ON f.id_fornecedor = c.id_fornecedor

UNION

SELECT c.id_compra, c.quantidade, c.data_compra,
       m.id_medicamento, m.nome, m.preco, m.validade, m.laboratorio,
       f.id_fornecedor, f.nome, f.cidade
FROM Fornecedor f
LEFT JOIN Compra c
    ON c.id_fornecedor = f.id_fornecedor
LEFT JOIN Medicamento m
    ON m.id_medicamento = c.id_medicamento;


/* =============================================================================
   6) CLÍNICA

    id_medico     INT PRIMARY KEY,

    id_consulta   INT PRIMARY KEY,
    id_paciente   INT NOT NULL,
