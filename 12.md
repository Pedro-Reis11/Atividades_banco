SELECT p.nome AS produto, COUNT(*) AS 'quantidade de filiais'
FROM Produto p
JOIN Estoque e 
ON p.id = e.id_produto
GROUP BY p.id, p.nome
HAVING COUNT(*) >= 1;

SELECT p.nome AS produto, SUM(e.quantidade) AS quantidade_total
FROM Produto p
JOIN Estoque e 
ON p.id = e.id_produto
GROUP BY p.id, p.nome
HAVING SUM(e.quantidade) > 100;

SELECT p.nome as produto, ROUND(AVG(e.preco), 2) as preco_medio, f.nome as filial
FROM Produto p
JOIN Estoque e 
ON p.id = e.id_produto
JOIN Filial f ON e.cnpj_filial = f.cnpj
GROUP BY p.nome,f.nome
ORDER BY preco_medio DESC;

SELECT p.nome as produto, MAX(e.preco) as maior_preco , MIN(e.preco) as menor_preco
FROM Produto p 
JOIN Estoque e 
ON p.id = e.id_produto
GROUP BY p.nome
HAVING MAX(e.preco) - MIN(e.preco) > 20.00;

SELECT f.nome as filial, SUM(e.quantidade) as estoque_total
FROM Filial f 
JOIN Estoque e 
ON f.cnpj = e.cnpj_filial
GROUP BY f.nome, f.cnpj
HAVING SUM(e.quantidade) > 500;

SELECT p.nome as produto, COUNT(*) as quantidade_filiais
FROM Produto p 
JOIN Estoque e 
ON p.id = e.id_produto
GROUP BY p.nome, p.id
HAVING COUNT(*) >= 3;

SELECT c.nome as cliente, COUNT(p.id_pedido) as pedidos
FROM Cliente c 
JOIN Pedido p 
ON c.id_cliente = p.id_cliente
GROUP BY c.nome, c.id_cliente
HAVING COUNT(p.id_pedido) >= 1;

SELECT c.nome as cliente, SUM(p.total) as pedidos
FROM Cliente c 
JOIN Pedido p 
ON c.id_cliente = p.id_cliente
GROUP BY c.nome, c.id_cliente
HAVING SUM(p.total) >= 500;

SELECT p.nome as produto, SUM(ip.quantidade) as quantidade_vendidas
FROM Produto p 
JOIN ItemPedido ip
ON p.id_produto = ip.id_produto
GROUP BY p.nome, p.id_produto
HAVING SUM(ip.quantidade) > 30;

SELECT c.nome AS cliente, AVG(p.total) AS valor_medio_pedido
FROM Cliente c 
JOIN Pedido p ON c.id_cliente = p.id_cliente
GROUP BY c.id_cliente, c.nome
HAVING AVG(p.total) > 80.00;

SELECT p.data_pedido, COUNT(p.id_pedido) AS total_pedidos
FROM Pedido p 
JOIN Cliente c ON p.id_cliente = c.id_cliente 
GROUP BY p.data_pedido 
HAVING COUNT(p.id_pedido) > 10;
